# ---------------------------------------------------------------------------
# DIY SystemCore: Raspberry Pi 5 + Waveshare "2-CH CAN HAT" (2x MCP2515 on SPI0)
#
# The stock image's five MCP2518FD buses (sc-mcp2518-can*-spi* overlays on
# SPI1/SPI2/SPI3) are REMOVED here: their pins (GPIO 8 = CAN4 INT, GPIO 9 =
# CAN2 INT, GPIO 24/25 = SPI2/SPI1 chip selects, ...) collide with SPI0.
#
# Waveshare 2-CH CAN HAT pinout (wiki table, unmodified board):
#   CAN_0: CS = GPIO8 (CE0) -> spi0.0   INT_0 = GPIO23 (or 22 if 0R pad moved)
#   CAN_1: CS = GPIO7 (CE1) -> spi0.1   INT_1 = GPIO25 (or 24 if 0R pad moved)
#   16 MHz crystal, MCP2515 max SPI clock 10 MHz.
# The udev rule 71-diy-can-interface-names.rules names spi0.0 -> can_s0 and
# spi0.1 -> can_s1 (CANBus.systemCore(0) / (1) in robot code).
# ---------------------------------------------------------------------------
dtparam=spi=on
dtoverlay=mcp2515-can0,oscillator=16000000,interrupt=23,spimaxfrequency=10000000
dtoverlay=mcp2515-can1,oscillator=16000000,interrupt=25,spimaxfrequency=10000000

#working gadget
dtoverlay=dwc2,dr_mode=peripheral
dtparam=pciex1_gen3

camera_auto_detect=0
display_auto_detect=0
dtparam=i2s=off
dtparam=i2c_csi_dsi0=on
# The stock image uses "dtoverlay=i2c1,pins_10_11". GPIO 10/11 are SPI0
# MOSI/SCLK on the 40-pin header, which the CAN HAT needs, and the RP1 SPI and
# I2C drivers are both modules so whichever probes second loses its pins.
# Keep i2c1 on its normal header pins (GPIO 2/3) instead.
dtoverlay=i2c1

uart_2ndstage=0
